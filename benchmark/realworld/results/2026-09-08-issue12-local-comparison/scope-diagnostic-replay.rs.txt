    #[test]
    #[ignore = "temporary saved-snapshot diagnosis"]
    fn replay_scope_diagnostic_snapshot() {
        let snapshot: serde_json::Value = serde_json::from_slice(
            &std::fs::read("/tmp/pdfdelta-sp800-scope-snapshot.json").unwrap(),
        ).unwrap();
        let number = |value: &serde_json::Value| usize::try_from(value.as_u64().unwrap()).unwrap();
        let make_blocks = |side: &str| snapshot[side].as_array().unwrap().iter().map(|value| {
            assert!(value["unmapped"].as_array().unwrap().is_empty());
            block(value["id"].as_u64().unwrap(), value["text"].as_str().unwrap())
        }).collect::<Vec<_>>();
        let old = make_blocks("old");
        let new = make_blocks("new");
        let make_range = |value: &serde_json::Value| ResolvedScopeRange {
            start: ScopeCoordinate { block_order: number(&value[0][0]), scalar: number(&value[0][1]) },
            end: ScopeCoordinate { block_order: number(&value[1][0]), scalar: number(&value[1][1]) },
        };
        let scopes = snapshot["scopes"].as_array().unwrap().iter().map(|value| ResolvedScope {
            id: value["id"].as_str().unwrap().to_owned(),
            old: make_range(&value["old"]),
            new: make_range(&value["new"]),
        }).collect::<Vec<_>>();
        let mut failed = 0;
        for (index, value) in snapshot["changes"].as_array().unwrap().iter().enumerate() {
            let occurrences = value["occurrences"].as_array().unwrap().iter().map(|value| {
                let span = |side: &str| {
                    let value = &value[side];
                    (!value.is_null()).then(|| TextSpan {
                        blocks: value["blocks"].as_array().unwrap().iter().map(|value| BlockId(value.as_u64().unwrap())).collect(),
                        separator: match value["separator"].as_str() {
                            Some("Space") => Some(BlockSeparator::Space),
                            Some("Concatenate") => Some(BlockSeparator::Concatenate),
                            None => None,
                            other => panic!("unknown separator {other:?}"),
                        },
                        canonical_range: ScalarRange { start: number(&value["canonical"][0]), end: number(&value["canonical"][1]) },
                        comparable_range: TokenRange { start: number(&value["comparable"][0]), end: number(&value["comparable"][1]) },
                    })
                };
                pdfdelta_core::diff::ChangeOccurrence { old_span: span("old"), new_span: span("new") }
            }).collect();
            let change = Change { kind: ChangeKind::Replacement, occurrences, confidence: Confidence::High, tags: Vec::new() };
            if let Err(error) = classify_scoped_changes(std::slice::from_ref(&change), &scopes, &old, &new) {
                failed += 1;
                eprintln!("FAILED change {index}: {error}");
                for occurrence in &change.occurrences {
                    for (side, span, blocks) in [("old", &occurrence.old_span, &old), ("new", &occurrence.new_span, &new)] {
                        let Some(span) = span else { continue };
                        let mut budget = ClassificationBudget::default();
                        let limits = ClassificationLimits::default();
                        let order = classification_block_order(blocks, &mut budget, limits).unwrap();
                        let range = span_range_with_limits(span, blocks, &order, &mut budget, limits);
                        let scope = range.as_ref().map(|range| containing_scope(*range, &scopes, side, &mut budget, limits));
                        eprintln!("  {side}: {span:?} range={range:?} scope={scope:?}");
                        for id in &span.blocks {
                            let block = &blocks[order[id]];
                            eprintln!("    block {} text={:?}", id.0, block.canonical.text);
                        }
                    }
                }
            }
        }
        eprintln!("FAILED TOTAL {failed}");
    }
