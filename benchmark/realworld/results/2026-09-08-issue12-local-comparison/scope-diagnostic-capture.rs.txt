    if let Ok(path) = std::env::var("PDFDELTA_SCOPE_DIAGNOSTIC") {
        let span_json = |span: &TextSpan| serde_json::json!({
            "blocks": span.blocks.iter().map(|block| block.0).collect::<Vec<_>>(),
            "separator": span.separator.map(|separator| format!("{separator:?}")),
            "canonical": [span.canonical_range.start, span.canonical_range.end],
            "comparable": [span.comparable_range.start, span.comparable_range.end],
        });
        let blocks_json = |blocks: &[BlockText]| blocks.iter().map(|block| serde_json::json!({
            "id": block.block.0,
            "text": block.canonical.text,
            "unmapped": block.canonical.unmapped.iter().map(|token| token.scalar_index).collect::<Vec<_>>(),
        })).collect::<Vec<_>>();
        let snapshot = serde_json::json!({
            "old": blocks_json(old_blocks),
            "new": blocks_json(new_blocks),
            "scopes": scopes.iter().map(|scope| serde_json::json!({
                "id": scope.id,
                "old": [[scope.old.start.block_order, scope.old.start.scalar], [scope.old.end.block_order, scope.old.end.scalar]],
                "new": [[scope.new.start.block_order, scope.new.start.scalar], [scope.new.end.block_order, scope.new.end.scalar]],
            })).collect::<Vec<_>>(),
            "changes": changes.iter().map(|change| serde_json::json!({
                "kind": format!("{:?}", change.kind),
                "occurrences": change.occurrences.iter().map(|occurrence| serde_json::json!({
                    "old": occurrence.old_span.as_ref().map(span_json),
                    "new": occurrence.new_span.as_ref().map(span_json),
                })).collect::<Vec<_>>(),
            })).collect::<Vec<_>>(),
        });
        std::fs::write(path, serde_json::to_vec(&snapshot).expect("diagnostic JSON"))
            .expect("diagnostic snapshot");
    }
