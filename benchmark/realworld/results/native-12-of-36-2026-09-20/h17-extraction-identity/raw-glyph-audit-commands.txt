#!/usr/bin/env bash
set -euo pipefail
# Executed complete raw-glyph audit, NASA new side, H16 full006 -> H17 full008.
ROOT=/home/hayato/ghq/github.com/hayatosc/pdfdelta
S=/tmp/opencode/h17-raw
CACHE="$ROOT/benchmark/realworld/cache/native-12-of-36-2026-09-20"
RES="$ROOT/benchmark/realworld/results/native-12-of-36-2026-09-20/h17-extraction-identity"
PDF="$ROOT/benchmark/realworld/cache/next-dev/nasa-buckling-8007-1968-to-2020-new.pdf"
rm -rf "$S"; mkdir -p "$S/h16" "$S/h17"
tar xzf "$CACHE/h16-full-iteration-006-native/production.tar.gz" -C "$S/h16"
tar xzf "$CACHE/h17-full-iteration-008-native/production.tar.gz" -C "$S/h17"
for ver in h16 h17; do cp "$RES/raw_glyph_stream.rs" "$S/$ver/crates/pdfdelta-bench/examples/raw_glyph_stream.rs"; done
for ver in h16 h17; do (cd "$S/$ver" && cargo build --release -p pdfdelta-bench --example raw_glyph_stream); done
"$S/h16/target/release/examples/raw_glyph_stream" "$PDF" 2>"$S/h16.run.err" | gzip > "$S/h16-new.jsonl.gz"
"$S/h17/target/release/examples/raw_glyph_stream" "$PDF" 2>"$S/h17.run.err" | gzip > "$S/h17-new.jsonl.gz"
python "$RES/raw_glyph_compare.py" \
 "$S/h16-new.jsonl.gz" "$S/h17-new.jsonl.gz" \
 "$CACHE/h16-full-iteration-006-native/production.tar.gz" \
 "$CACHE/h17-full-iteration-008-native/production.tar.gz" \
 "$PDF" "$RES/raw-glyph-audit-nasa.json" "$RES/raw-glyph-audit-failed-prefix.json"
